from datetime import datetime
from typing import Optional
from uuid import UUID
from pydantic import field_validator
from sqlmodel import Field, SQLModel, Column, JSON
from app.db import uuid7

class WebRTCSettingsBase(SQLModel):
    enabled: bool = Field(default=False)
    auth_mode: str = Field(default="extension", max_length=32, description="extension|webrtc|both")
    wss_url: Optional[str] = Field(default=None, max_length=256)
    sip_domain: Optional[str] = Field(default=None, max_length=128)
    ice_servers: Optional[list] = Field(default=None, sa_column=Column(JSON))

    @field_validator("auth_mode")
    @classmethod
    def validate_auth_mode(cls, v: str) -> str:
        if v not in ("extension", "webrtc", "both"):
            raise ValueError("auth_mode must be extension|webrtc|both")
        return v

    @field_validator("wss_url")
    @classmethod
    def validate_wss_url(cls, v: Optional[str]) -> Optional[str]:
        if v is None:
            return v
        if not (v.startswith("wss://") or v.startswith("ws://")):
            raise ValueError("wss_url must start with wss:// or ws://")
        return v

class WebRTCSettings(WebRTCSettingsBase, table=True):

    caller_domain = getattr(user, "_domain_id", user.domain_id)

        return {"domain_id": str(domain_id), "enabled": False, "auth_mode": "extension", "wss_url": None, "sip_domain": None, "ice_servers": None}
    return {"domain_id": str(row.domain_id), "enabled": row.enabled, "auth_mode": row.auth_mode, "wss_url": row.wss_url, "sip_domain": row.sip_domain, "ice_servers": row.ice_servers, "updated_at": row.updated_at.isoformat()}

@router.put("/{domain_id}/webrtc_settings")
async def put_webrtc_settings(domain_id: UUID, payload: dict, session: AsyncSession = Depends(get_session), user: User = Depends(require_permission("domain_edit"))):
    caller_domain = getattr(user, "_domain_id", user.domain_id)
    # allow superadmin to edit any domain; else must match
    # validation via model
    from app.webrtc_settings.models import WebRTCSettingsBase
    try:
        validated = WebRTCSettingsBase(**payload)
    except Exception as e:
        raise HTTPException(status_code=422, detail=str(e))
    await set_tenant(session, domain_id)
    r = await session.execute(select(WebRTCSettings).where(WebRTCSettings.domain_id == domain_id))
    row = r.scalar_one_or_none()
    if row:
        row.enabled = validated.enabled
        row.auth_mode = validated.auth_mode
        row.wss_url = validated.wss_url
        row.sip_domain = validated.sip_domain
        row.ice_servers = validated.ice_servers
        row.updated_at = datetime.utcnow()
    else:
        row = WebRTCSettings(domain_id=domain_id, enabled=validated.enabled, auth_mode=validated.auth_mode, wss_url=validated.wss_url, sip_domain=validated.sip_domain, ice_servers=validated.ice_servers)
        session.add(row)
    await session.commit()
    await session.refresh(row)
    return {"domain_id": str(row.domain_id), "enabled": row.enabled, "auth_mode": row.auth_mode, "wss_url": row.wss_url, "sip_domain": row.sip_domain, "ice_servers": row.ice_servers}

# app/webrtc/credentials_router.py
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from app.auth.deps import get_current_user
from app.auth.models import User
from app.db import get_session, set_tenant

router = APIRouter(prefix="/api/v1/webrtc", tags=["webrtc"])

@router.post("/credentials")
async def get_credentials(payload: dict, session: AsyncSession = Depends(get_session), user: User = Depends(get_current_user)):
    domain_id = getattr(user, "_domain_id", user.domain_id)
    await set_tenant(session, domain_id)
    from app.webrtc_settings.models import WebRTCSettings
    from app.extensions.models import Extension
    from app.domains.models import Domain
    r = await session.execute(select(WebRTCSettings).where(WebRTCSettings.domain_id == domain_id))
    settings = r.scalar_one_or_none()
    if not settings or not settings.enabled:
        raise HTTPException(status_code=400, detail="WebPhone disabled for this domain")
    auth_mode = settings.auth_mode
    wss_url = settings.wss_url
    if not wss_url:
        raise HTTPException(status_code=400, detail="wss_url not configured")
    # sip_domain fallback
    sip_domain = settings.sip_domain
    if not sip_domain:
        r2 = await session.execute(select(Domain).where(Domain.domain_id == domain_id))
        dom = r2.scalar_one_or_none()
        sip_domain = dom.domain_name if dom else "example.com"
    # handle extension mode
    ext_val = payload.get("extension")
    webrtc_user = payload.get("webrtc_user")
    webrtc_pass = payload.get("webrtc_pass")
    if ext_val:
        if auth_mode not in ("extension", "both"):
            raise HTTPException(status_code=400, detail="extension auth disabled")
        r3 = await session.execute(select(Extension).where(Extension.domain_id == domain_id, Extension.extension == str(ext_val)))
        ext = r3.scalar_one_or_none()
        if not ext or not ext.enabled:
            raise HTTPException(status_code=404, detail="Extension not found")
        # optional: check user owns extension later; phase1 allow any in domain
        return {"sip_uri": f"sip:{ext.extension}@{sip_domain}", "wss_url": wss_url, "auth_user": str(ext.extension), "password": ext.password or "1234", "display_name": ext.effective_caller_id_name or str(ext.extension), "expires_in": 3600, "ice_servers": settings.ice_servers or [{"urls": "stun:stun.l.google.com:19302"}]}
    elif webrtc_user:
        if auth_mode not in ("webrtc", "both"):
            raise HTTPException(status_code=400, detail="webrtc auth disabled")
        # phase1 echo

    else:
        raise HTTPException(status_code=422, detail="Provide extension or webrtc_user")

  const regRef=React.useRef<any>(null);
  const sessionRef=React.useRef<any>(null);

  const storageKey=React.useMemo(()=>{
    try{ const token=localStorage.getItem("token"); if(!token) return null; const payload=JSON.parse(atob(token.split(".")[1])); return `webphone:cfg:${payload.domain_id}:${payload.sub}`; }catch{return null}
  },[]);

  // hydrate
  React.useEffect(()=>{
    if(!storageKey) return;
    const raw=localStorage.getItem(storageKey);
    if(raw){ try{ const cfg=JSON.parse(raw); setConfig(cfg); setRegistrationState("unregistered"); }catch{} }
  },[storageKey]);

  // auto-connect if config + enabled
  React.useEffect(()=>{
    if(!config) return;
    // fetch settings to ensure enabled
    const domainId = localStorage.getItem("domain_id") || (()=>{
      try{ return JSON.parse(atob(localStorage.getItem("token")!.split(".")[1])).domain_id }catch{return null}
    })();
    if(!domainId) return;
    apiFetch(`/domains/${domainId}/webrtc_settings`).then((s:any)=>{
      if(s.enabled) connect(config);
    }).catch(()=>{});
    // eslint-disable-next-line
  },[config?.sip_uri]);

  async function connect(cfg:WebPhoneConfig){
    setRegistrationState("connecting"); setRegistrationError(null);
    try{
      const { UserAgent, Registerer } = await import("sip.js");
      const uri = UserAgent.makeURI(cfg.sip_uri);
      if(!uri) throw new Error("Invalid SIP URI");
      const ua = new UserAgent({
        uri,
        transportOptions: { server: cfg.wss_url, iceServers: cfg.ice_servers },
        authorizationUsername: cfg.auth_user,
        authorizationPassword: cfg.password,
        displayName: cfg.display_name,
      } as any);
      ua.delegate = {
        onInvite: (inv:any)=>{
          sessionRef.current=inv;
          setCallState("incoming");
          inv.stateChange.addListener((st:any)=>{ if(st==="Established") setCallState("confirmed"); if(st==="Terminated") {setCallState("idle"); sessionRef.current=null;} });
        }
      };
      await ua.start();
      const registerer = new Registerer(ua);
      // @ts-ignore
      registerer.stateChange.addListener((state:string)=>{
        if(state==="Registered") setRegistrationState("registered");
        else if(state==="Unregistered") setRegistrationState("unregistered");
        else if(state==="Terminated") setRegistrationState("failed");
      });
      await registerer.register();
      uaRef.current=ua; regRef.current=registerer;
      if(storageKey) localStorage.setItem(storageKey, JSON.stringify(cfg));
      setConfig(cfg);
      setRegistrationState("registered");
    }catch(e:any){
      setRegistrationState("failed"); setRegistrationError(e.message?.slice(0,300) || "connect failed");
    }
  }

  async function disconnect(){
    try{ await regRef.current?.unregister(); await uaRef.current?.stop(); }catch{}
    uaRef.current=null; regRef.current=null;
    setRegistrationState("unconfigured"); setCallState("idle");
    if(storageKey) localStorage.removeItem(storageKey);
    setConfig(null);
  }
  async function call(dest:string){
    if(!uaRef.current) throw new Error("not registered");
    const { Inviter, SessionState } = await import("sip.js");
    const uri = (await import("sip.js")).UserAgent.makeURI(`sip:${dest}@${config?.sip_uri.split("@")[1]}`);
    if(!uri) throw new Error("bad dest");
    const inv = new Inviter(uaRef.current, uri, { sessionDescriptionHandlerOptions: { constraints: {audio:true, video:false}}});
    sessionRef.current=inv;
    setCallState("trying");
    inv.stateChange.addListener((s:string)=>{
      if(s===SessionState.Establishing) setCallState("proceeding");
      else if(s===SessionState.Established) setCallState("confirmed");
      else if(s===SessionState.Terminated){ setCallState("idle"); sessionRef.current=null; }
    });
    await inv.invite();
  }
  async function hangup(){ try{ await sessionRef.current?.bye(); }catch{} setCallState("idle"); sessionRef.current=null; }
  async function answer(){ try{ await sessionRef.current?.accept(); setCallState("confirmed"); }catch{} }
  async function reject(){ try{ await sessionRef.current?.reject(); }catch{} setCallState("idle"); sessionRef.current=null; }
  function mute(v:boolean){ try{ const pc=sessionRef.current?.sessionDescriptionHandler?.peerConnection; pc?.getSenders().forEach((s:any)=>{ if(s.track?.kind==="audio") s.track.enabled=!v; }); }catch{} setIsMuted(v); }
  function hold(v:boolean){ setIsHeld(v); /* sip.js hold via invite with a=inactive - phase2 */ }
  function sendDTMF(tone:string){ try{ sessionRef.current?.sessionDescriptionHandler?.sendDtmf(tone); }catch{} }

  // duration ticker
  React.useEffect(()=>{ if(callState!=="confirmed") return; const id=setInterval(()=>setDuration(d=>d+1),1000); return()=>clearInterval(id); },[callState]);

  return <WebPhoneContext.Provider value={{registrationState, registrationError, callState, config, isMuted, isHeld, connect, disconnect, call, hangup, answer, reject, mute, hold, sendDTMF, duration}}>{children}</WebPhoneContext.Provider>;
}
